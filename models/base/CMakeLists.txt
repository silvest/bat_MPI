set(headers BCFitter.h BCEfficiencyFitter.h BCGraphFitter.h BCHistogramFitter.h)
set(deps BAT::BAT)
set(linkdefs "${CMAKE_CURRENT_SOURCE_DIR}/LDef.h")
if(BAT_ENABLE_ROOSTATS)
  list(APPEND headers BCRooInterface.h BATCalculator.h)
  list(APPEND deps ROOT::RooFit ROOT::RooFitCore ROOT::RooStats)
  list(APPEND linkdefs "${CMAKE_CURRENT_SOURCE_DIR}/LDef_roostats.h")
endif()
set(sources)
foreach(header IN LISTS headers)
  string(REPLACE ".h" ".cxx" source "${header}")
  list(APPEND sources "${source}")
endforeach()
list(TRANSFORM headers PREPEND "${CMAKE_CURRENT_SOURCE_DIR}/")
bat_add_library(BATmodels HEADERS ${headers} SOURCES ${sources} LINKDEF ${linkdefs} DEPENDENCIES ${deps})
