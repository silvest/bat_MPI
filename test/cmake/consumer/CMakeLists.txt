cmake_minimum_required(VERSION 3.16)
project(BATInstalledConsumer LANGUAGES CXX)
find_package(BAT 1 CONFIG REQUIRED COMPONENTS PT)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE BAT::BAT)
if(TARGET BAT::BATmodels)
  target_link_libraries(consumer PRIVATE BAT::BATmodels)
  target_compile_definitions(consumer PRIVATE TEST_MODELS)
endif()
if(TARGET BAT::BATmtf)
  target_link_libraries(consumer PRIVATE BAT::BATmtf)
  target_compile_definitions(consumer PRIVATE TEST_MTF)
endif()
if(BAT_STATIC_CONSUMER AND TARGET BAT::BAT_static)
  set_property(TARGET consumer PROPERTY LINK_LIBRARIES BAT::BAT_static)
  if(TARGET BAT::BATmodels_static)
    target_link_libraries(consumer PRIVATE BAT::BATmodels_static)
  endif()
  if(TARGET BAT::BATmtf_static)
    target_link_libraries(consumer PRIVATE BAT::BATmtf_static)
  endif()
endif()
# Installed downstream executables also need a runtime search path.
set_property(TARGET consumer PROPERTY INSTALL_RPATH "${BAT_LIBRARY_DIR}")
install(TARGETS consumer RUNTIME DESTINATION bin)
