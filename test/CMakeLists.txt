if(NOT BAT_BUILD_MODELS OR NOT BAT_BUILD_MTF)
  message(FATAL_ERROR "Legacy tests require BAT_BUILD_MODELS and BAT_BUILD_MTF")
endif()
add_library(BAT_test_support STATIC test.cxx GaussModel.cxx IntegrationModel.cxx)
target_include_directories(BAT_test_support PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_BINARY_DIR}/src" "${PROJECT_SOURCE_DIR}")
target_link_libraries(BAT_test_support PUBLIC BAT::BAT BAT::BATmodels BAT::BATmtf)
foreach(case IN ITEMS test BCAux BCEngineMCMC BCFitter BCHistogram BCMath BCModel BCParameter BCPrior BCSummaryTool parallel BCIntegrate)
  add_executable(${case}_legacy_TEST ${case}_TEST.cxx)
  target_link_libraries(${case}_legacy_TEST PRIVATE BAT_test_support)
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${case}")
  add_test(NAME legacy_${case} COMMAND ${case}_legacy_TEST)
  set_tests_properties(legacy_${case} PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${case}")
endforeach()
