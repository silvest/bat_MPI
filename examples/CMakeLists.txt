# The original standalone examples run on one MPI process. ROOT .C macros
# remain available as source examples, but are not executable CMake targets.
set(example_dirs basic/poisson basic/binomial basic/errorPropagation basic/tutorial
  basic/combination1d basic/combination2d basic/rootOutput advanced/trialFunction)
if(BAT_BUILD_MODELS)
  list(APPEND example_dirs advanced/referenceCounting advanced/polynomialFit)
endif()
if(BAT_BUILD_MTF)
  list(APPEND example_dirs advanced/mtf/systematics)
endif()
if(BAT_ENABLE_ROOSTATS)
  list(APPEND example_dirs advanced/rooInterface)
endif()
foreach(folder IN LISTS example_dirs)
  file(GLOB sources CONFIGURE_DEPENDS "${folder}/*.cxx")
  set(mains)
  set(models)
  foreach(source IN LISTS sources)
    file(READ "${source}" contents)
    if(contents MATCHES "int main\\(")
      list(APPEND mains "${source}")
    else()
      list(APPEND models "${source}")
    endif()
  endforeach()
  foreach(main IN LISTS mains)
    get_filename_component(base "${main}" NAME_WE)
    string(REPLACE "/" "_" target "example_${folder}_${base}")
    add_executable(${target} "${main}" ${models})
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(${target} PRIVATE BAT::BAT)
    foreach(lib IN ITEMS BATmodels BATmtf)
      if(TARGET BAT::${lib})
        target_link_libraries(${target} PRIVATE BAT::${lib})
      endif()
    endforeach()
  endforeach()
endforeach()
